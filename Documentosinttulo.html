const LS_KEY = "miDia_v1";

let datos = JSON.parse(localStorage.getItem(LS_KEY) || "{}");

let reporteRango = "semana";

const hoyISO = () => new Date().toISOString().slice(0,10);

const getDia = (f) => { if(!datos[f]) datos[f]={}; return datos[f]; };

const guardar = () => localStorage.setItem(LS_KEY, JSON.stringify(datos));

function setValor(fecha, itemId, valor, sat, nota){

  const d = getDia(fecha);

  if(valor === null || valor === undefined || valor === false || valor === ""){

    if(sat === undefined && nota === undefined) delete d[itemId];

    else d[itemId] = {v:false, s:sat, n:nota, t:Date.now()};

  } else {

    d[itemId] = {v:valor, s:sat!==undefined?sat:(d[itemId]?.s||null), n:nota!==undefined?nota:(d[itemId]?.n||""), t:Date.now()};

  }

  guardar();

}

const getValor = (f,id) => datos[f]?.[id]?.v;

const getSat = (f,id) => datos[f]?.[id]?.s;

const getNota = (f,id) => datos[f]?.[id]?.n || "";

function toast(msg){

  const t = document.getElementById("toast");

  t.textContent = msg; t.classList.add("show");

  setTimeout(() => t.classList.remove("show"), 2000);

}

function showView(id, btn){

  document.querySelectorAll(".view").forEach(v => v.classList.remove("active"));

  document.querySelectorAll("nav button").forEach(b => b.classList.remove("active"));

  document.getElementById("view-" + id).classList.add("active");

  btn.classList.add("active");

  if(id === "hoy") renderHoy();

  if(id === "esporadicas") renderEsporadicas();

  if(id === "informes") renderInformes();

}

function crearElementoItem(item, fecha){

  const v = getValor(fecha, item.id);

  const sat = getSat(fecha, item.id);

  const n = getNota(fecha, item.id);

 

  const container = document.createElement("div");

  const wrap = document.createElement("div");

  wrap.className = "item" + (v ? " done" : "");

 

  const label = document.createElement("div");

  label.className = "label";

  label.innerHTML = `${item.label}${item.hint?`<span class="sub">${item.hint}</span>`:""}`;

  wrap.appendChild(label);

  if(item.tipo === "toggle"){

    const t = document.createElement("div");

    t.className = "toggle" + (v ? " on" : "");

    t.onclick = () => {

      setValor(fecha, item.id, !getValor(fecha, item.id));

      renderHoy(); // Rápido re-render

    };

    wrap.appendChild(t);

  } else if(item.tipo === "scale"){

    const s = document.createElement("div"); s.className = "scale";

    for(let i=0; i<=10; i++){

      const b = document.createElement("button");

      b.textContent = i; b.dataset.v = i;

      if(v === i) b.classList.add("on");

      b.onclick = () => {

        setValor(fecha, item.id, v === i ? null : i);

        renderHoy();

      };

      s.appendChild(b);

    }

    wrap.appendChild(s);

  } else if(item.tipo === "stepper"){

    const s = document.createElement("div"); s.className = "stepper";

    const btnMinus = document.createElement("button"); btnMinus.textContent = "−";

    const valSpan = document.createElement("div"); valSpan.className = "val"; valSpan.textContent = v || 0;

    const btnPlus = document.createElement("button"); btnPlus.textContent = "+";

    btnMinus.onclick = () => { setValor(fecha, item.id, Math.max(0, (v||0) - item.step)); renderHoy(); };

    btnPlus.onclick = () => { setValor(fecha, item.id, Math.min(item.max, (v||0) + item.step)); renderHoy(); };

    s.appendChild(btnMinus); s.appendChild(valSpan); s.appendChild(btnPlus);

    wrap.appendChild(s);

  } else if(item.tipo === "chips"){

    const s = document.createElement("div"); s.className = "chips";

    item.opts.forEach(op => {

      const c = document.createElement("button"); c.className = "chip" + (v === op ? " on" : "");

      c.textContent = op;

      c.onclick = () => { setValor(fecha, item.id, v === op ? null : op); renderHoy(); };

      s.appendChild(c);

    });

    wrap.appendChild(s);

  }

  container.appendChild(wrap);

  // Fila de satisfacción (solo si está activo o si es tipo escala)

  if(v || item.tipo === "scale"){

    const meta = document.createElement("div");

    meta.className = "meta-row";

    const ml = document.createElement("span"); ml.className = "meta-label"; ml.textContent = "Satisfacción";

    meta.appendChild(ml);

    const ss = document.createElement("div"); ss.className = "scale";

    for(let i=0; i<=10; i++){

      const b = document.createElement("button");

      b.textContent = i; b.dataset.v = i;

      if(sat === i) b.classList.add("on");

      b.onclick = () => { setValor(fecha, item.id, v, sat === i ? null : i, n); renderHoy(); };

      ss.appendChild(b);

    }

    meta.appendChild(ss);

    container.appendChild(meta);

  }

  // Nota (solo si está activo)

  if(v && item.tipo !== "scale"){

    const nr = document.createElement("div"); nr.className = "note-row";

    const inp = document.createElement("input");

    inp.placeholder = "Nota rápida...";

    inp.value = n;

    inp.onchange = (e) => { setValor(fecha, item.id, v, sat, e.target.value); };

    nr.appendChild(inp);

    container.appendChild(nr);

  }

  return container;

}

function renderSeccion(idSec, contenedor){

  const sec = CATALOGO[idSec];

  if(!sec) return;

  const fecha = hoyISO();

  const div = document.createElement("div"); div.className = "section";

  div.innerHTML = `<h2 class="section-title"><span class="em">${sec.em}</span>${sec.titulo}</h2>`;

  const card = document.createElement("div"); card.className = "card";

  sec.items.forEach(it => card.appendChild(crearElementoItem(it, fecha)));

  div.appendChild(card);

  contenedor.appendChild(div);

}

function renderHoy(){

  const c = document.getElementById("hoyContainer");

  c.innerHTML = "";

  Object.keys(CATALOGO).forEach(k => {

    if(CATALOGO[k].enHoy) renderSeccion(k, c);

  });

  actualizarProgreso();

}

function renderEsporadicas(){

  const c = document.getElementById("esporadicasContainer");

  c.innerHTML = "";

  Object.keys(CATALOGO).forEach(k => {

    if(!CATALOGO[k].enHoy) renderSeccion(k, c);

  });

}

function actualizarProgreso(){

  const fecha = hoyISO();

  let total = 0, hechos = 0;

  Object.values(CATALOGO).forEach(sec => {

    if(sec.enHoy){

      sec.items.forEach(it => {

        if(it.tipo === "toggle"){ total++; if(getValor(fecha, it.id)) hechos++; }

      });

    }

  });

  const pct = total === 0 ? 0 : Math.round((hechos/total)*100);

  document.getElementById("progressRing").style.setProperty("--p", pct + "%");

  document.getElementById("progressTxt").textContent = pct + "%";

}

function marcarCasaRapido(){

  renderHoy();

  toast("Casa completada ✨");

}

function renderInformes(){

  const c = document.getElementById("informeContainer");

  c.innerHTML = "";

  const fechas = Object.keys(datos).sort().reverse();

  let filtradas = fechas;

  const hoy = new Date();

  if(reporteRango === "semana"){ const lim = new Date(); lim.setDate(hoy.getDate()-7); filtradas = fechas.filter(f => new Date(f) >= lim); }

  else if(reporteRango === "mes"){ const lim = new Date(); lim.setDate(hoy.getDate()-30); filtradas = fechas.filter(f => new Date(f) >= lim); }

 

  if(filtradas.length === 0){

    c.innerHTML = '<div class="chart-empty">No hay datos en este periodo.</div>';

    return;

  }

  // Resumen general

  const card = document.createElement("div"); card.className = "report-card";

  card.innerHTML = `<h3>📅 Resumen de ${filtradas.length} días</h3>`;

  let totalChecks = 0;

  filtradas.forEach(f => { totalChecks += Object.keys(datos[f]).filter(k => datos[f][k].v).length; });

  card.innerHTML += `<div class="kv"><span>Actividades registradas</span><b>${totalChecks}</b></div>`;

  c.appendChild(card);

  // Gráfico de dolor (Canvas nativo)

  const canvas = document.createElement("canvas");

  canvas.width = 600; canvas.height = 300;

  const ctx = canvas.getContext("2d");

  const dolores = ["d-espalda","d-rodillas","d-manos","e-ansiedad"];

 

  let minD = Infinity, maxD = -Infinity;

  const pts = dolores.map(d => {

    return filtradas.slice(0,7).reverse().map(f => {

      const v = datos[f]?.[d]?.v;

      if(typeof v === "number"){ minD=Math.min(minD,v); maxD=Math.max(maxD,v); return {f, v}; }

      return {f, v:null};

    });

  });

  if(minD === Infinity) minD = 0; if(maxD === -Infinity) maxD = 10;

  if(minD === maxD) maxD = minD + 1;

  ctx.fillStyle = "#fff"; ctx.fillRect(0,0,600,300);

  ctx.strokeStyle = "#eee"; ctx.lineWidth = 1;

  for(let i=0; i<=10; i++){

    const y = 280 - (i/10)*240;

    ctx.beginPath(); ctx.moveTo(40, y); ctx.lineTo(580, y); ctx.stroke();

    ctx.fillStyle = "#888"; ctx.font = "12px sans-serif"; ctx.fillText(i, 15, y+4);

  }

  const colors = ["#e07a5f","#7c6bb0","#4caf7a","#e0b03c"];

  const n = pts[0].length;

  const stepX = n > 1 ? 540 / (n-1) : 0;

  pts.forEach((arr, idx) => {

    ctx.strokeStyle = colors[idx]; ctx.lineWidth = 3; ctx.beginPath();

    let started = false;

    arr.forEach((p, i) => {

      if(p.v !== null){

        const x = 40 + i*stepX;

        const y = 280 - ((p.v - minD)/(maxD-minD))*240;

        if(!started){ ctx.moveTo(x,y); started=true; } else { ctx.lineTo(x,y); }

      }

    });

    ctx.stroke();

   

    // Leyenda

    ctx.fillStyle = colors[idx]; ctx.fillRect(40+idx*130, 10, 12, 12);

    ctx.fillStyle = "#333"; ctx.font = "bold 12px sans-serif";

    ctx.fillText(dolores[idx].replace("d-","").replace("e-",""), 58+idx*130, 21);

  });

  const gcard = document.createElement("div"); gcard.className = "report-card";

  gcard.innerHTML = "<h3>📈 Evolución Dolor y Ánimo</h3>";

  gcard.appendChild(canvas);

  c.appendChild(gcard);

}

function importarDatos(e){

  const file = e.target.files[0]; if(!file) return;

  const reader = new FileReader();

  reader.onload = (ev) => {

    try{ datos = JSON.parse(ev.target.result); guardar(); renderHoy(); toast("Importado"); }

    catch(err){ alert("Error al leer archivo"); }

  };

  reader.readAsText(file);

}

}
